# The library the repository is named after: representation, encoding, parsing
# and validation of GxNet telegrams.
#
# No dependencies, and no transport. Everything that talks to a device lives in
# link/; everything a person clicks lives in app/.

add_library(gxnet
    src/core.cpp
    src/escape.cpp
    src/telegram.cpp
    src/binary.cpp
    src/codec.cpp
    src/validate.cpp
    src/gxnet_c.cpp
)

target_include_directories(gxnet
    PUBLIC  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
            $<INSTALL_INTERFACE:include>
    PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src
)

# The optional table of subfunction meanings, generated by tools/gen_docs.py and
# not part of the repository. registry.hpp finds it on its own with
# __has_include, which is what a build without CMake relies on -- but a header
# that does not exist yet leaves no trace in any depfile, so generating it into
# an existing build tree would rebuild nothing and the program would go on
# reporting that it knows no meanings. Stating it on the command line instead
# gives the build system something that changes when the file appears.
#
# A glob rather than EXISTS, over a path with no wildcard in it: CONFIGURE_DEPENDS
# is the one mechanism that checks at build time whether a file has come into
# existence. Naming the file directly would only notice it disappearing, which
# is the less useful half. Ninja and the Makefile generators honour it; the
# others re-check when CMake is next run.
file(GLOB GXNET_TOKEN_DOCS CONFIGURE_DEPENDS
     ${CMAKE_CURRENT_SOURCE_DIR}/include/gxnet/detail/registry_docs.hpp)
if(GXNET_TOKEN_DOCS)
    target_compile_definitions(gxnet PUBLIC GXNET_HAS_TOKEN_DOCS=1)
    message(STATUS "gxnet: subfunction meanings found")
else()
    message(STATUS "gxnet: no subfunction meanings; generate them with tools/gen_docs.py")
endif()

if(MSVC)
    # /utf-8 sets both the source and the execution character set. Without it
    # MSVC reads sources as the system code page and narrows string literals to
    # it, which on a Russian Windows quietly mangles any non-ASCII text on its
    # way to a label. Everything here is UTF-8 end to end.
    target_compile_options(gxnet PRIVATE /W4 /utf-8)
else()
    target_compile_options(gxnet PRIVATE -Wall -Wextra -Wpedantic)
endif()

# Included here rather than relying on the top level: this directory is added
# before the top-level include(GNUInstallDirs) runs.
include(GNUInstallDirs)
install(TARGETS gxnet ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
